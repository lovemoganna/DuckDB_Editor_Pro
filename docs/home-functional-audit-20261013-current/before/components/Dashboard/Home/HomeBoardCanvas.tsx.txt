import React, { useCallback, useEffect, useRef, useState } from 'react';
import { BarChart3, LayoutDashboard, Lock, Unlock, X, SquareTerminal, Table2, ChevronRight } from 'lucide-react';
import { Dashboard, Tab } from '../../../types';
import { DashboardGrid } from '../../DashboardGrid';
import { dbService } from '../../../services/dbService';
import { toastService } from '../../../services/toastService';
import { DB } from '../dashboardUi';
import type { BoardBrushFilter } from './homeBoardTypes';
import { WorkspaceObjects } from './WorkspaceObjects';
import { DashboardContextControls, DatasetComposer, DatasetWidgetBuilder } from './WorkbenchControls';
import { DatasetPreview } from './DatasetPreview';
import { useAppStore } from '../../../hooks/store/useAppStore';
import { compileAnalysis, analysisSqlForEditor } from '../../../services/workbench/queryCompiler';
import { loadDatasets } from '../../../services/workbench/datasets';
import { ActionButton, FormSelect } from '../../ui/Workbench';
import { useWorkbenchRevision } from '../../../services/workbench/runtime';
import type { DatasetDefinition, WidgetSettings } from '../../../services/workbench/analysisTypes';

export type { BoardBrushFilter } from './homeBoardTypes';

export interface HomeBoardCanvasProps {
  dashboard: Dashboard | null;
  loading?: boolean;
  refreshTrigger: number;
  /** Optional SQL {{param}} substitution; UI entry removed until widgets use placeholders. */
  boardParams?: Record<string, string>;
  brushFilter: BoardBrushFilter | null;
  onBrushFilterChange: (filter: BoardBrushFilter | null) => void;
  onDashboardChange: (dashboard: Dashboard) => void;
  onNavigate: (tab: Tab) => void;
  onGenerateChart?: () => void;
  onNewQuery?: () => void;
  onSelectTable?: (tableName: string) => void;
  onSelectTableStructure?: (tableName: string) => void;
  onSelectTableDataFlow?: (tableName: string) => void;
}

export const HomeBoardCanvas: React.FC<HomeBoardCanvasProps> = ({
  dashboard,
  loading = false,
  refreshTrigger,
  boardParams = {},
  brushFilter,
  onBrushFilterChange,
  onDashboardChange,
  onNavigate,
  onGenerateChart,
  onNewQuery,
  onSelectTable,
  onSelectTableStructure,
  onSelectTableDataFlow,
}) => {
  const revision = useWorkbenchRevision();
  const [datasets,setDatasets] = useState<DatasetDefinition[]>([]);
  const [selectedDataset, setSelectedDataset] = useState('');
  const activeDataset = useAppStore(s => s.activeDataset);
  const [datasetError,setDatasetError] = useState('');
  const [loadedOwner,setLoadedOwner] = useState('');
  const [timerRevision,setTimerRevision] = useState(0);
  const [builderOpen, setBuilderOpen] = useState(false);
  const layoutSaveInFlight = useRef(false);
  const [layoutSaving, setLayoutSaving] = useState(false);
  const owner = dashboard?.databaseName || 'memory';
  const currentView = useRef({ id: dashboard?.id, owner });
  currentView.current = { id: dashboard?.id, owner };
  const publish = useCallback((next: Dashboard) => {
    if (currentView.current.id === next.id && currentView.current.owner === next.databaseName) onDashboardChange(next);
    window.dispatchEvent(new CustomEvent('duckdb-home-dashboard-updated'));
  }, [onDashboardChange]);
  const datasetGeneration = useRef(0);
  const refreshDatasets = useCallback(async () => {
    const generation = ++datasetGeneration.current;
    try {
      const next = await loadDatasets(owner);
      if (generation !== datasetGeneration.current) return;
      setDatasets(next);
      setLoadedOwner(owner);
      setDatasetError('');
    } catch (error) {
      if (generation === datasetGeneration.current) {
        setDatasetError(`数据集读取失败：${String((error as Error).message)}`);
        // Allow dashboard to render even when dataset listing fails
        setLoadedOwner(prev => prev || owner);
      }
    }
  }, [owner]);
  useEffect(() => {
    void refreshDatasets();
    return () => { datasetGeneration.current += 1; };
  }, [refreshDatasets, revision]);
  useEffect(() => {
    const current=useAppStore.getState().activeDataset;
    setSelectedDataset(current?.catalog===owner && !current.id.startsWith('sql:') ? current.id : '');
  }, [owner]);
  useEffect(() => {
    if (loadedOwner === owner && activeDataset && datasets.some(d => d.id === activeDataset.id)) setSelectedDataset(activeDataset.id);
  }, [loadedOwner, owner, activeDataset?.id, datasets]);
  useEffect(() => {
    if (loadedOwner !== owner || !selectedDataset) return;
    const dataset = datasets.find(item => item.id === selectedDataset);
    if (!dataset) { setSelectedDataset(''); useAppStore.getState().setActiveDataset(null); return; }
    try {
      const compiled = compileAnalysis({sql:'',catalog:dataset.catalog,schema:dataset.schema,datasets,
        globalFilters:dashboard?.context?.filters,spec:{datasetId:dataset.id,dimensions:[],aggregation:'DETAIL'}});
      useAppStore.getState().setActiveDataset({...dataset,sql:analysisSqlForEditor(compiled),origin:'home',appliedFilters:dashboard?.context?.filters || []});
    } catch(error) { useAppStore.getState().setActiveDataset({...dataset,error:String((error as Error).message)}); }
  }, [selectedDataset, datasets, loadedOwner, owner, dashboard?.context]);
  useEffect(()=>{const seconds=dashboard?.context?.refreshSeconds;if(!seconds)return;const timer=window.setInterval(()=>setTimerRevision(n=>n+1),Math.max(30,seconds)*1000);return()=>window.clearInterval(timer);},[dashboard?.id,dashboard?.context?.refreshSeconds]);
  const saveBoard = useCallback(async(next:Dashboard)=>{
    const updated = await dbService.updateDashboard(next.id, next.databaseName!, current => ({...current,formatVersion:2,context:next.context}));
    publish(updated);
  },[publish]);
  const saveSettings = useCallback(async(id:string,settings:WidgetSettings)=>{
    if(!dashboard)return;
    const updated = await dbService.updateDashboard(dashboard.id, owner, current => {
      if (!current.items.some(item => item.i === id)) throw new Error('此组件已移除，请重新打开视图');
      return {...current,formatVersion:2,items:current.items.map(item=>item.i===id?{...item,settings}:item)};
    });
    publish(updated);
  },[dashboard?.id,owner,publish]);
  /** Default locked: window resize must not rewrite saved x/y/w/h */
  const layoutLocked = dashboard ? dashboard.layoutLocked !== false : true;

  const handleLayoutChange = useCallback(
    async (layout: Array<{ i: string; x: number; y: number; w: number; h: number }>) => {
      if (!dashboard || dashboard.layoutLocked !== false) return;
      try {
        const updated = await dbService.updateDashboard(dashboard.id, owner, current => ({ ...current, items: current.layoutLocked !== false ? current.items : current.items.map((item) => {
        const l = layout.find((x) => x.i === item.i);
        if (!l) return item;
        return { ...item, x: l.x, y: l.y, w: l.w, h: l.h };
        }) }));
        publish(updated);
      } catch(error) { toastService.error(`布局未保存，原布局已保留：${String((error as Error).message)}`); }
    },
    [dashboard, owner, publish],
  );

  const handleToggleLayoutLock = useCallback(async () => {
    if (!dashboard || layoutSaveInFlight.current) return;
    layoutSaveInFlight.current = true;
    setLayoutSaving(true);
    try {
      const updated = await dbService.updateDashboard(dashboard.id, owner, current => ({...current,layoutLocked:current.layoutLocked === false}));
      publish(updated);
      toastService.success(updated.layoutLocked ? '布局已锁定，窗口缩放不再移动图表' : '布局已解锁，可拖拽与调整尺寸');
    } catch(error) { toastService.error(`布局状态未保存：${String((error as Error).message)}`); }
    finally { layoutSaveInFlight.current = false; setLayoutSaving(false); }
  }, [dashboard?.id, owner, publish]);

  const handleRemoveWidget = useCallback(
    async (itemId: string) => {
      if (!dashboard) return;
      try {
        const updated = await dbService.updateDashboard(dashboard.id, owner, current => ({...current,items:current.items.filter(i=>i.i!==itemId)}));
        publish(updated);
      } catch(error) { toastService.error(`组件未移除：${String((error as Error).message)}`); }
    },
    [dashboard?.id, owner, publish],
  );

  const isEmpty = !dashboard || dashboard.items.length === 0;
  const initialLoading = loading && !dashboard;
  const resultRevision = refreshTrigger + timerRevision + revision;

  return (
    <section aria-busy={loading} className="relative z-0 flex flex-col gap-3">
      <WorkspaceObjects
        board={dashboard}
        onChange={onDashboardChange}
        onCreateChart={() => setBuilderOpen(true)}
        controls={<>
          {dashboard && (
            <DatasetWidgetBuilder
              board={dashboard}
              datasets={datasets}
              initialDatasetId={selectedDataset}
              brush={brushFilter}
              open={builderOpen}
              onOpenChange={setBuilderOpen}
              onCreated={publish}
            />
          )}
          {onNewQuery && (
            <ActionButton size="sm" variant="secondary" onClick={onNewQuery} className="text-monokai-cyan hover:text-monokai-cyan">
              <SquareTerminal className="h-3.5 w-3.5" aria-hidden="true" />
              <span>新建查询</span>
            </ActionButton>
          )}
          <label className="flex items-center gap-1.5 text-xs text-monokai-fg-muted font-medium">
            <span>数据集</span>
            <FormSelect
              sizeVariant="sm"
              aria-label="浏览数据集"
              className="!w-auto min-w-[140px] max-w-[240px]"
              value={selectedDataset}
              onChange={event => {setSelectedDataset(event.target.value); if(!event.target.value)useAppStore.getState().setActiveDataset(null);}}
            >
              <option value="">选择数据集</option>
              {datasets.map(dataset => <option key={dataset.id} value={dataset.id}>{dataset.name}</option>)}
            </FormSelect>
          </label>
          {dashboard && <DashboardContextControls board={dashboard} datasets={datasets} onSave={saveBoard}/>}
        </>}
        createControls={dashboard && <div className="flex flex-wrap gap-2"><DatasetComposer owner={owner} datasets={datasets} onCreated={refreshDatasets}/></div>}
        settings={dashboard?.description && (
          <details className="w-full text-xs text-monokai-fg-muted rounded-md border border-monokai-border bg-monokai-surface/40 p-2">
            <summary className="cursor-pointer font-medium text-monokai-fg hover:text-monokai-cyan">指标口径说明</summary>
            <p className="mt-1.5 leading-relaxed text-monokai-fg-muted">{dashboard.description}</p>
          </details>
        )}
      />
      {datasetError && (
        <div role="alert" className="flex items-center justify-between rounded-md border border-monokai-pink/30 bg-monokai-pink/10 px-3 py-2 text-xs text-monokai-pink">
          <p>{datasetError}</p>
          <button type="button" aria-label="重试读取数据集" className="underline hover:text-monokai-pink/80 cursor-pointer font-medium ml-2" onClick={() => void refreshDatasets()}>
            重试读取数据集
          </button>
        </div>
      )}
      <div className="flex flex-wrap items-center justify-between gap-2 pt-1 border-t border-monokai-border/40">
        <span className="text-2xs font-mono text-monokai-comment">
          {dashboard?.items.length ? `${dashboard.items.length} 项分析组件` : '尚未添加分析组件'}
        </span>
        <div className="flex items-center gap-1.5">
          {brushFilter && (
            <button
              type="button"
              onClick={() => onBrushFilterChange(null)}
              className={`${DB.chip} hover:border-monokai-pink/40 hover:text-monokai-pink cursor-pointer transition-colors`}
              title="清除联动筛选"
            >
              <span>筛选: {brushFilter.column}={brushFilter.value}</span>
              <X className="h-3 w-3" />
            </button>
          )}
          {!isEmpty && (
            <ActionButton
              size="sm"
              variant={layoutLocked ? 'secondary' : 'primary'}
              disabled={layoutSaving}
              loading={layoutSaving}
              onClick={() => void handleToggleLayoutLock()}
              title={
                layoutLocked
                  ? '解锁后可拖拽并调整每个图表宽高；窗口缩放不会改变布局'
                  : '锁定后固定所有图表位置与尺寸'
              }
            >
              {layoutLocked ? (
                <>
                  <Unlock className="h-3.5 w-3.5" aria-hidden="true" />
                  <span>编辑布局</span>
                </>
              ) : (
                <>
                  <Lock className="h-3.5 w-3.5" aria-hidden="true" />
                  <span>完成编辑</span>
                </>
              )}
            </ActionButton>
          )}
        </div>
      </div>

      {initialLoading && (
        <div className={`${DB.empty} py-16`}>
          <span className="text-monokai-comment font-mono text-xs">加载看板…</span>
        </div>
      )}

      {!initialLoading && isEmpty && !selectedDataset && (
        <section className="rounded-lg border border-monokai-border/80 bg-monokai-surface/40 p-5 mt-1">
          <div className="flex items-center justify-between pb-3 border-b border-monokai-border/60">
            <div>
              <h2 className="text-sm font-semibold text-monokai-fg">开始分析</h2>
              <p className="text-2xs text-monokai-comment mt-0.5">选择已连接的数据集探索明细、生成聚合图表或构建自定义分析</p>
            </div>
            <span className="text-2xs font-mono text-monokai-comment">{datasets.length} 个可用数据集</span>
          </div>
          <div className="mt-3 grid max-h-72 grid-cols-1 gap-2 overflow-y-auto sm:grid-cols-2 custom-scrollbar pr-1">
            {datasets.map(dataset => (
              <button
                key={dataset.id}
                type="button"
                aria-label={dataset.name}
                title={dataset.name}
                onClick={() => setSelectedDataset(dataset.id)}
                className="group flex min-w-0 items-center justify-between gap-3 rounded-md border border-monokai-border/70 bg-monokai-surface/60 p-2.5 text-left transition-all hover:border-monokai-cyan/50 hover:bg-monokai-surface hover:shadow-xs focus-visible:outline focus-visible:outline-monokai-cyan"
              >
                <div className="flex min-w-0 items-center gap-2.5 flex-1">
                  <div className="flex h-7 w-7 shrink-0 items-center justify-center rounded border border-monokai-border bg-monokai-elevated/60 text-monokai-cyan group-hover:border-monokai-cyan/40">
                    <Table2 size={15} aria-hidden="true"/>
                  </div>
                  <div className="min-w-0 flex-1">
                    <span className="block truncate font-mono text-xs text-monokai-fg font-medium group-hover:text-monokai-cyan transition-colors">
                      {dataset.table || dataset.name}
                    </span>
                    <span className="block truncate text-2xs text-monokai-comment">
                      {dataset.schema} · {dataset.fields.length} 个字段
                    </span>
                  </div>
                </div>
                <ChevronRight size={14} className="shrink-0 text-monokai-comment group-hover:text-monokai-cyan transition-colors" aria-hidden="true"/>
              </button>
            ))}
            {datasets.length === 0 && (
              <div className="col-span-full py-8 text-center text-xs text-monokai-comment">
                暂无连接的数据集，请在上方导入数据或创建新表。
              </div>
            )}
          </div>
        </section>
      )}
      {!initialLoading && selectedDataset && loadedOwner === owner && (
        <DatasetPreview
          key={`${owner}:${selectedDataset}`}
          dataset={datasets.find(dataset => dataset.id === selectedDataset)}
          datasets={datasets}
          context={dashboard?.context}
          revision={resultRevision}
          onBuildChart={id => { setSelectedDataset(id); setBuilderOpen(true); }}
          onOpenGraph={() => onNavigate(Tab.ONTOLOGY)}
        />
      )}

      {!initialLoading && dashboard && loadedOwner===owner && dashboard.items.length > 0 && (
        <div
          className="relative z-0 overflow-x-auto py-1"
        >
          {!layoutLocked && (
            <p className="px-2 pb-2 text-2xs text-monokai-comment font-mono">
              拖拽标题栏移动 · 拖动边角调整宽高 · 完成后点「完成编辑」
            </p>
          )}
          <DashboardGrid
            dashboard={dashboard}
            refreshTrigger={resultRevision}
            datasets={datasets}
            context={dashboard.context}
            onSettingsChange={saveSettings}
            onLayoutChange={handleLayoutChange}
            onRemoveWidget={handleRemoveWidget}
            onNavigate={onNavigate}
            boardParams={boardParams}
            brushFilter={brushFilter}
            onBrushFilter={onBrushFilterChange}
            onSelectTable={onSelectTable}
            onSelectTableStructure={onSelectTableStructure}
            onSelectTableDataFlow={onSelectTableDataFlow}
          />
        </div>
      )}
    </section>
  );
};
