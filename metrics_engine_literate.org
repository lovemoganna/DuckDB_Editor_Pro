** 1.1 业务目标与文学工程定位
本工作流基于 =org-literate-engineering= 范式构建，将分散的 SQL 分析脚本转化为*可执行、可参数化、可复用、可验证、自带断言闭环*的文学编程系统。

** 1.2 核心约束与执行契约 (Invariant Contract)

- **固定逻辑与变量解耦**：汇率对齐与清洗逻辑沉淀于底层 CTE；高频变动的阈值（如 Top N、渠道过滤、最低 KYC 等）通过 =:var= 暴露给用户。

* 3. Reusable Metric Calculation Blocks

** 3.1 基础公共模型：统一 USDT 计价交易流 (Unified Ledger)
为了避免每个指标重复书写多币种汇率换算，构建统一标准化视图：

#+name: create_unified_views
#+begin_src sql
CREATE OR REPLACE VIEW v_unified_transactions AS
WITH fiat_norm AS (
    SELECT 
        f.fiat_order_id AS tx_id,
        CAST(f.uid AS BIGINT) AS uid,
        'fiat' AS asset_class,
        f.direction,
        f.currency AS asset_symbol,
        f.amount AS raw_amount,
        f.fee AS raw_fee,
        fx.rate_to_usdt,
        ROUND(f.amount * fx.rate_to_usdt, 4) AS amount_usdt,
        ROUND(f.fee * fx.rate_to_usdt, 4) AS fee_usdt,
        f.channel AS route,
        f.status,
        f.created_at,
        f.completed_at,
        ROUND(epoch(f.completed_at - f.created_at) / 60.0, 2) AS latency_mins
    FROM fiat_transactions f
    JOIN exchange_rates fx ON f.currency = fx.asset
),
crypto_norm AS (
    SELECT 
        c.crypto_order_id AS tx_id,
        CAST(c.uid AS BIGINT) AS uid,
        'crypto' AS asset_class,
        c.direction,
        c.asset AS asset_symbol,
        c.amount AS raw_amount,
        c.fee AS raw_fee,
        fx.rate_to_usdt,
        ROUND(c.amount * fx.rate_to_usdt, 4) AS amount_usdt,
        ROUND(c.fee * fx.rate_to_usdt, 4) AS fee_usdt,
        c.network AS route,
        c.status,
        c.created_at,
        c.completed_at,
        ROUND(epoch(c.completed_at - c.created_at) / 60.0, 2) AS latency_mins
    FROM crypto_transactions c
    JOIN exchange_rates fx ON c.asset = fx.asset
)
SELECT * FROM fiat_norm
UNION ALL
SELECT * FROM crypto_norm;
#+end_src

    ROUND(SUM(CASE WHEN direction = 'deposit' THEN amount_usdt ELSE 0 END), 2) AS gross_inflow_usdt,
    ROUND(SUM(CASE WHEN direction = 'withdrawal' THEN amount_usdt ELSE 0 END), 2) AS gross_outflow_usdt,
    ROUND(SUM(CASE WHEN direction = 'deposit' THEN amount_usdt ELSE -amount_usdt END), 2) AS net_funding_usdt,

    COUNT(*) AS completed_order_count
FROM v_unified_transactions
WHERE status = 'completed';

** 3.5 模块四：可参数化用户价值排行与大客户集中度 (Whale Ranking)

#+name: metric_user_ranking
#+begin_src sql :var top_limit=5
WITH user_aggregates AS (
    SELECT 
        u.uid,
        u.user_name,
        u.country_code,
        u.kyc_level,
        u.account_status,
        ROUND(SUM(CASE WHEN t.direction = 'deposit' AND t.status = 'completed' THEN t.amount_usdt ELSE 0 END), 2) AS deposit_usdt,
        ROUND(SUM(CASE WHEN t.direction = 'withdrawal' AND t.status = 'completed' THEN t.amount_usdt ELSE 0 END), 2) AS withdrawal_usdt,
        ROUND(SUM(CASE WHEN t.status = 'completed' THEN (CASE WHEN t.direction = 'deposit' THEN t.amount_usdt ELSE -t.amount_usdt END) ELSE 0 END), 2) AS net_balance_usdt,
        ROUND(SUM(CASE WHEN t.status = 'completed' THEN t.fee_usdt ELSE 0 END), 2) AS fee_paid_usdt,
        COUNT(CASE WHEN t.status = 'completed' THEN 1 END) AS completed_tx_count
    FROM users u
    LEFT JOIN v_unified_transactions t ON CAST(u.uid AS BIGINT) = t.uid
    GROUP BY u.uid, u.user_name, u.country_code, u.kyc_level, u.account_status
),
total_metrics AS (
    SELECT SUM(deposit_usdt) AS platform_total_deposit FROM user_aggregates
)
SELECT 
    a.uid,
    a.user_name,
    a.country_code,
    a.kyc_level,
    a.account_status,
    a.deposit_usdt,
    a.withdrawal_usdt,
    a.net_balance_usdt,
    a.fee_paid_usdt,
    a.completed_tx_count,

* 4. Verification & Consistency Assertions (Golden Contract)

本节运行数学与业务层面的守恒断言。在文学工程中，“执行成功 ≠ 结果正确”，断言是防止回归的关键。

#+name: verify_system_invariants
#+begin_src sql
WITH platform_kpi AS (
    SELECT ROUND(SUM(CASE WHEN direction = 'deposit' THEN amount_usdt ELSE -amount_usdt END), 2) AS platform_net
    FROM v_unified_transactions WHERE status = 'completed'
),
user_sum AS (
    SELECT ROUND(SUM(CASE WHEN t.direction = 'deposit' THEN t.amount_usdt ELSE -t.amount_usdt END), 2) AS users_net_sum
    FROM v_unified_transactions t WHERE t.status = 'completed'
),
tx_count AS (
    SELECT 
        COUNT(CASE WHEN asset_class = 'fiat' AND status = 'completed' THEN 1 END) AS fiat_ok,
        COUNT(CASE WHEN asset_class = 'crypto' AND status = 'completed' THEN 1 END) AS crypto_ok
    FROM v_unified_transactions
)
SELECT 
    p.platform_net,
    u.users_net_sum,
    (p.platform_net = u.users_net_sum) AS is_reconciliation_zero_variance,
    t.fiat_ok = 21 AS is_fiat_count_verified,
    t.crypto_ok = 21 AS is_crypto_count_verified,
    CASE 
        WHEN p.platform_net = u.users_net_sum AND t.fiat_ok = 21 AND t.crypto_ok = 21 
        THEN 'ALL INVARIANTS PASSED (PASS)'
        ELSE 'ASSERTION FAILED (FAIL)'
    END AS final_system_verdict
FROM platform_kpi p
CROSS JOIN user_sum u
CROSS JOIN tx_count t;
#+end_src

* 6. Diagnostic & Self-Repair Playbook

当未来接入新币种、发生汇率缺失或通道扩展时：

1. **未匹配币种诊断**：
   #+begin_src sql
   SELECT DISTINCT currency FROM fiat_transactions WHERE currency NOT IN (SELECT asset FROM exchange_rates)
   UNION
   SELECT DISTINCT asset FROM crypto_transactions WHERE asset NOT IN (SELECT asset FROM exchange_rates);
   #+end_src
   若返回非空行，立即在 =exchange_rates= 中补充基准汇率。

2. **清算延迟极端值诊断**：
   #+begin_src sql
   SELECT tx_id, route, latency_mins, raw_amount, asset_symbol 
   FROM v_unified_transactions 
   WHERE latency_mins > 600 AND status = 'completed';
   #+end_src
   可直接捕获 ACH 等超长清算通道的具体阻滞订单。
